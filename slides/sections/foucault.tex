\begin{frame}{Matrix factorisation by gradient descent}
  \begin{itemize}
    \item Model: $\hat r_{ui} = \mu + b_u + b_i + p_u^\top q_i$
      \begin{itemize}
        \item $\mu$: mean rating; $b_u$, $b_i$: user and movie biases;
        \item $p_u, q_i \in \mathbb{R}^k$: latent tastes of user $u$ and profile of movie $i$.
      \end{itemize}
    \item Loss on the observed cells $\Omega$:
      \[
        L = \sum_{(u,i) \in \Omega} (r_{ui} - \hat r_{ui})^2 + \lambda \left( \|P\|^2 + \|Q\|^2 \right) + \lambda_b \left( \|b_{\text{user}}\|^2 + \|b_{\text{item}}\|^2 \right)
      \]
    \item Mini-batch stochastic gradient descent, with $e_{ui} = r_{ui} - \hat r_{ui}$ and step $\eta$:
      \[
        p_u \leftarrow p_u + \eta \left( e_{ui}\, q_i - \tfrac{\lambda}{n_u}\, p_u \right),
        \qquad
        q_i \leftarrow q_i + \eta \left( e_{ui}\, p_u - \tfrac{\lambda}{n_i}\, q_i \right)
      \]
    \item Mini-batches rather than full batch: one step size cannot suit a user with 1\,000 ratings and one with 1.
    \item Hyperparameters $k$, $\lambda$, $\lambda_b$, $\eta$ and the number of epochs, tuned on the 5 folds.
  \end{itemize}
\end{frame}
