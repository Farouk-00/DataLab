\begin{frame}{Matrix factorisation by gradient descent}
  \begin{itemize}
    \item Model: $\hat r_{ui} = \mu + b_u + b_i + p_u^\top q_i$, with $p_u, q_i \in \mathbb{R}^k$.
    \item Loss on the observed cells $\Omega$:
      \[
        \sum_{(u,i) \in \Omega} (r_{ui} - \hat r_{ui})^2 + \lambda \left( \|P\|^2 + \|Q\|^2 \right) + \lambda_b \left( \|b_{\text{user}}\|^2 + \|b_{\text{item}}\|^2 \right)
      \]
    \item Mini-batch SGD (256 ratings), learning rate $\eta$, $\mathcal{N}(0, 0.1)$ initialisation.
    \item Full-batch steps are dominated by heavy users: one $\eta$ cannot suit a user with 1\,000 ratings and one with 1.
    \item The penalty is spread as $\lambda / n_u$ over each rating of $u$, so $\lambda$ means the same as in ALS.
    \item Tuned on the 5 folds: $k$, $\lambda$, $\lambda_b$, $\eta$, number of epochs.
  \end{itemize}
\end{frame}
