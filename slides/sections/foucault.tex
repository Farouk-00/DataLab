\begin{frame}{Matrix factorisation by gradient descent}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \begin{itemize}
        \item Model: $\hat r_{ui} = \mu + b_u + b_i + p_u^\top q_i$, with $p_u, q_i \in \mathbb{R}^k$.
        \item Loss on the observed cells $\Omega$:
          \[
            \sum_{\Omega} (r_{ui} - \hat r_{ui})^2 + \lambda \left( \|P\|^2 + \|Q\|^2 \right) + \lambda_b \|b\|^2
          \]
        \item Mini-batch SGD (256 ratings): full-batch steps are dominated by users with many ratings.
        \item Chosen on the 5 folds: $k = 10$, $\lambda = 10$, $\lambda_b = 3$, $\eta = 0.02$, 80 epochs.
        \item 5-fold RMSE $0.882 \pm 0.007$ (ALS $0.888$, biases $0.884$); test RMSE $0.874$; 1.8\,s.
        \item Without $p_u^\top q_i$: $0.883$. The factors barely help at 1\,\% density.
      \end{itemize}
    \end{column}

      \centering
      \includegraphics[width=\linewidth]{gd/rmse_epochs.pdf}\\
      {\footnotesize Small $\lambda$ overfits after 7 epochs.}
    \end{column}
  \end{columns}
\end{frame}
