\begin{frame}{Matrix factorisation by gradient descent (AdamW)}
  \small
  \begin{itemize}
    \item Model: $\hat r_{ui} = \mu + b_u + b_i + p_u^\top q_i$
      \begin{itemize}
        \item $\mu$: mean rating; $b_u$, $b_i$: user and movie biases;
        \item $p_u, q_i \in \mathbb{R}^k$: latent tastes of user $u$ and profile of movie $i$, vectors of size $k$.
      \end{itemize}
    \item Loss on the observed cells $\Omega$, the same as ALS:
      \[
        L = \sum_{(u,i) \in \Omega} (r_{ui} - \hat r_{ui})^2 + \lambda \left( \|P\|^2 + \|Q\|^2 \right) + \lambda_b \left( \|b_{\text{user}}\|^2 + \|b_{\text{item}}\|^2 \right)
      \]
    \item Full-batch gradient $g = \nabla_\theta L$, AdamW step for each parameter $\theta$:
      \[
        m \leftarrow \beta_1 m + (1 - \beta_1)\, g, \qquad
        v \leftarrow \beta_2 v + (1 - \beta_2)\, g^2, \qquad
        \theta \leftarrow \theta - \eta \left( \frac{\hat m}{\sqrt{\hat v} + \varepsilon} + w\, \theta \right)
      \]
    \item The step adapts to each parameter: users with 1\,000 ratings and with 1 learn at the same pace.
    \item $\lambda$ stays in the loss: the decay $w$ shrinks a movie rated once as much as one rated 300 times.
    \item Hyperparameters $k$, $\lambda$, $\lambda_b$, $\eta$, $w$ and the number of epochs, tuned on the 5 folds.
  \end{itemize}
\end{frame}
